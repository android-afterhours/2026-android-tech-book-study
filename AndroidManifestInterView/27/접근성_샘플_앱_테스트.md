```kotlin
/**
 * 접근성 트리 순서는 좌 컬럼(1,3) → 우 컬럼(2,4) 이라 시각적 순서와 어긋난다.
 * "직전에 읽을 뷰"를 지정해 행 우선으로 되돌린다. nextFocus* 와는 완전히 별개라
 * 위쪽 토글을 아무리 바꿔도 이 순서는 영향을 받지 않는다.
 */
private fun applyTraversalOrder(visual: Boolean) {
    val (et1, et2, et3, et4) = fields()
    if (visual) {
        et2.accessibilityTraversalAfter = et1.id
        et3.accessibilityTraversalAfter = et2.id
        et4.accessibilityTraversalAfter = et3.id
    } else {
        fields().forEach { it.accessibilityTraversalAfter = View.NO_ID }
    }
    ...
}
```

## 4. 색상 대비

### 코드

`ContrastUtil.kt` — WCAG 2.1 대비비 계산

```kotlin
/**
 * https://www.w3.org/TR/WCAG21/#dfn-contrast-ratio
 * 밝은 쪽 상대휘도를 L1, 어두운 쪽을 L2라 할 때 (L1 + 0.05) / (L2 + 0.05).
 */
fun contrastRatio(foreground: Int, background: Int): Double {
    val a = relativeLuminance(foreground)
    val b = relativeLuminance(background)
    val lighter = maxOf(a, b)
    val darker = minOf(a, b)
    return (lighter + 0.05) / (darker + 0.05)
}

/** https://www.w3.org/TR/WCAG21/#dfn-relative-luminance */
fun relativeLuminance(color: Int): Double =
    0.2126 * linearize(Color.red(color)) +
        0.7152 * linearize(Color.green(color)) +
        0.0722 * linearize(Color.blue(color))

/** sRGB 감마 보정을 되돌려 선형 광량으로 변환 */
private fun linearize(channel: Int): Double {
    val s = channel / 255.0
    return if (s <= 0.03928) s / 12.92 else ((s + 0.055) / 1.055).pow(2.4)
}
```

`section/ColorContrastActivity.kt` — 색을 뷰에서 직접 읽어 화면에 표시

```kotlin
/**
 * 색상 상수를 따로 두지 않고 뷰에서 직접 읽는다 — XML 색을 바꾸면 표시값이 자동으로 따라온다.
 */
private fun renderContrast(sample: TextView, result: TextView) {
    val foreground = sample.currentTextColor
    val background = (sample.background as? ColorDrawable)?.color ?: Color.WHITE
    val ratio = ContrastUtil.contrastRatio(foreground, background)
    val verdict = when {
        ratio >= ContrastUtil.AAA_NORMAL_TEXT -> R.string.a11y_s4_verdict_aaa
        ratio >= ContrastUtil.AA_NORMAL_TEXT -> R.string.a11y_s4_verdict_aa
        else -> R.string.a11y_s4_verdict_fail
    }
    result.text = getString(R.string.a11y_s4_ratio_format, ratio, getString(verdict))
}
```

## 5. 커스텀 뷰 + AccessibilityDelegate

### 코드

`CircleToggleView.kt` — `View` 를 직접 상속한 원형 토글. 접근성 정보가 하나도 자동으로 채워지지 않는다.

```kotlin
init {
    // 터치와 D-패드 양쪽에서 도달 가능해야 발화 차이를 비교할 수 있다
    isClickable = true
    isFocusable = true
}

/**
 * 클릭 리스너를 내부에서 소비하지 않고 performClick 을 재정의해야
 * 외부 리스너와 접근성 클릭 이벤트가 모두 정상 동작한다.
 */
override fun performClick(): Boolean {
    isOn = !isOn
    return super.performClick()
}

override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
    val desired = (DEFAULT_SIZE_DP * resources.displayMetrics.density).toInt()
    setMeasuredDimension(
        resolveSize(desired, widthMeasureSpec),
        resolveSize(desired, heightMeasureSpec)
    )
}
```

`section/CustomViewDelegateActivity.kt` — 같은 뷰 2개 중 **왼쪽에만** 델리게이트를 붙인다.

```kotlin
// 좋은 예: 커스텀 뷰가 스스로 알리지 못하는 역할·레이블·동작을 델리게이트가 채워준다
ViewCompat.setAccessibilityDelegate(
    binding.ctvWithDelegate,
    object : AccessibilityDelegateCompat() {
        override fun onInitializeAccessibilityNodeInfo(
            host: View,
            info: AccessibilityNodeInfoCompat
        ) {
            super.onInitializeAccessibilityNodeInfo(host, info)
            // 역할을 알려야 TalkBack 이 "버튼"으로 읽고 조작 힌트를 덧붙인다
            info.className = Button::class.java.name
            info.text = getString(toggleLabelRes(binding.ctvWithDelegate.isOn))
            info.addAction(AccessibilityNodeInfoCompat.AccessibilityActionCompat.ACTION_CLICK)
        }
    }
)

// 상태가 바뀌었음을 알리지 않으면 TalkBack 이 직전 레이블을 계속 읽는다
binding.ctvWithDelegate.setOnClickListener {
    it.sendAccessibilityEvent(AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED)
}

// 나쁜 예: binding.ctvWithoutDelegate 는 일부러 델리게이트를 붙이지 않아
//         역할도 레이블도 없이 읽힌다. 발화 차이를 비교하는 것이 목적.
```
